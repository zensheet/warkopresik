Contoh Apps Script

Buat Google Sheets bernama "Database Warkop Resik" dengan tab "Transaksi", lalu tempel kode ini di Extensions › Apps Script dan deploy sebagai Web App (akses: Anyone).

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName('Transaksi');
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.timestamp,
    data.id,
    data.customer,
    data.items,
    data.total
  ]);
  return ContentService.createTextOutput('OK');
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName('Transaksi');
  var lastRow = sheet.getLastRow();
  var limit = 100;
  var startRow = Math.max(2, lastRow - limit + 1);
  var numRows = lastRow - startRow + 1;
  var rows = numRows > 0
    ? sheet.getRange(startRow, 1, numRows, 5).getValues()
    : [];
  var result = rows.map(function(r) {
    return {
      timestamp: r[0],
      id: r[1],
      customer: r[2],
      items: r[3],
      total: r[4]
    };
  }).reverse();
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}
